203bool AgendaContactos::cargar(const string& nombre_fichero){
204
205 ifstream fichero(nombre_fichero);
206 if (!fichero) return false;
207
208 // Si hiciese falta limpiar la Agenda
209 // listaContactos.clear();
210 // etiquetas.clear();
211
212 string linea;
213
214 while (getline(fichero, linea)) {
215
216 string nombre, telefonos, correos, tags;
217 stringstream ss(linea);
218
219 getline(ss, nombre, '|');
220 getline(ss, telefonos, '|');
221 getline(ss, correos, '|');
222 getline(ss, tags, '|');
223
224 Contacto nuevoContacto = Contacto();
225
226 nuevoContacto.setNombre(nombre);
227
228 string dato;
229
230 stringstream ssTel(telefonos);
231 while ( getline(ssTel, dato, ',' )) {
232 if (!dato.empty()) nuevoContacto.añadirTelefono(dato);
233 }
234
235 stringstream ssCor(correos);
236 while (getline(ssCor, dato, ',')) {
237 if (!dato.empty()) nuevoContacto.añadirCorreo(dato);
238 }
239
240
241 stringstream ssTag(tags);
242 while (getline(ssTag, dato, ',')) {
243 if (!dato.empty()) nuevoContacto.añadirEtiqueta(dato);
244
245 }
246
247 if( !this->insertarContacto(nuevoContacto) ){
248 return false;
249 }
250 }
251
252 return true;

255bool AgendaContactos::guardar(const string& nombre_fichero ) const{
256
257 ofstream fichero(nombre_fichero);

259
260 for (auto it = listaContactos.begin(); it != listaContactos.end(); ++it) {

272 ++sig;
273 if (sig != telefonos.end()) fichero << ",";
274 }
275 fichero << "|";
276
277 // Correos
278 const set<string>& correos = c.getCorreos();
279 for (auto itCor = correos.begin(); itCor != correos.end(); ++itCor) {
280 fichero << *itCor;
281 auto sig = itCor;
282 ++sig;
283 if (sig != correos.end()) fichero << ",";
284 }
285 fichero << "|";
286
287 // Etiquetas
288 const set<string>& etiquetasContacto = c.getEtiquetas();
289 for (auto itEt = etiquetasContacto.begin(); itEt != etiquetasContacto.end(); ++itEt) {
290 fichero << *itEt;
291 auto sig = itEt;
292 ++sig;
293 if (sig != etiquetasContacto.end()) fichero << ",";
